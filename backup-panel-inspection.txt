<script id="rc-qrcode-script">
(function(){
  function createReceiptAuth(){
    const codeEl = document.getElementById('receiptAuthCode');
    const qrEl = document.getElementById('receiptQrCode');
    if(!codeEl || !qrEl || typeof QRCode === 'undefined') return;

    const stamp = new Date();
    const code = 'RC-' +
      stamp.getFullYear() + '-' +
      String(stamp.getMonth()+1).padStart(2,'0') +
      String(stamp.getDate()).padStart(2,'0') + '-' +
      Math.floor(Math.random()*9000+1000);

    codeEl.textContent = code;

    qrEl.innerHTML = '';

    new QRCode(qrEl,{
      text:'ReciboCondo|' + code + '|Emitido em ' + stamp.toLocaleString('pt-BR'),
      width:76,
      height:76
    });
  }

  if(document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', createReceiptAuth);
  }else{
    createReceiptAuth();
  }
})();
</script>
